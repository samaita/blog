{{- /* Project-level override of PaperMod's `templates/opengraph.html`. The theme emits exactly one `og:image`, and its lookup order ends at the site-wide fallback before it ever considers images that live inside the rendered post body. It also only understands a `cover` front matter block, not `featured_image`. This partial replaces that lookup so a post resolves its OG image in this order: 1. `featured_image` (or `images`) from the page front matter. 2. The first image that appears in the post content. 3. The site's default OG image / logo. Priority 2 prefers Hugo's resource API: an image that is a page resource resolves to its fingerprinted `Permalink`. For remote or statically-linked images (e.g. `/images/benchmark.png` under `static/`) there is no resource, so the first Markdown/HTML image reference is lifted out of the content. That extraction is deliberately conservative: it runs only as a fallback, it is never used when a front matter image exists, and it reports no image rather than guessing when it cannot parse a link. */ -}} {{- /* Resolve a single image reference into an absolute URL. */ -}} {{- define "partials/templates/og-image-url.html" -}} {{- $img := . -}} {{- $url := urls.Parse $img -}} {{- if ne $url.Scheme "" -}} {{- /* Absolute external URL (http, https, data, ...) — keep verbatim. */ -}} {{- $img -}} {{- else -}} {{- /* Relative site URL — must become absolute for OG consumers. */ -}} {{- $img | absURL -}} {{- end -}} {{- end -}} {{- $ogImage := "" -}} {{- /* 1. Featured image from page front matter. */ -}} {{- $frontMatterImage := "" -}} {{- with .Params.featured_image -}} {{- if reflect.IsSlice . -}} {{- $frontMatterImage = index . 0 -}} {{- else -}} {{- $frontMatterImage = . -}} {{- end -}} {{- end -}} {{- if not $frontMatterImage -}} {{- with .Params.images -}} {{- if reflect.IsSlice . -}} {{- $frontMatterImage = index . 0 -}} {{- else -}} {{- $frontMatterImage = . -}} {{- end -}} {{- end -}} {{- end -}} {{- with $frontMatterImage -}} {{- $ogImage = partial "templates/og-image-url.html" . -}} {{- end -}} {{- /* 2. First image in the post content. */ -}} {{- if not $ogImage -}} {{- $resourceImage := "" -}} {{- /* Prefer a real image resource when Hugo has one. */ -}} {{- with .Resources.ByType "image" -}} {{- with index . 0 -}} {{- $resourceImage = .Permalink -}} {{- end -}} {{- end -}} {{- if $resourceImage -}} {{- $ogImage = $resourceImage -}} {{- else -}} {{- /* No resource: pull the first image link out of the rendered content. Resource-less images are usually remote URLs or files under `static/`, and both cases carry the full path in the markup. */ -}} {{- $content := .Content -}} {{- $candidates := slice -}} {{- range $pattern := slice `(?is)]*?src=["']([^"']+)["'][^>]*>` `(?is)srcset=["']([^\s,"']+)` `(?is)]*?src=["']([^"']+)["'][^>]*>` -}} {{- with findRE $pattern $content -}} {{- range . -}} {{- with replaceRE $pattern "$1" . -}} {{- $candidates = $candidates | append (trim . " \t\n\r") -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $candidate := $candidates -}} {{- if not $ogImage -}} {{- $lower := lower $candidate -}} {{- /* Skip non-content images (theme icons, analytics beacons, ...). */ -}} {{- if and (not (strings.HasPrefix $lower "data:")) (not (strings.HasPrefix $lower "blob:")) -}} {{- if not (or (strings.Contains $lower "favicon") (strings.Contains $lower "avatar")) -}} {{- $url := urls.Parse $candidate -}} {{- if or (ne $url.Scheme "") (strings.HasSuffix $url.Path ".png") (strings.HasSuffix $url.Path ".jpg") (strings.HasSuffix $url.Path ".jpeg") (strings.HasSuffix $url.Path ".webp") (strings.HasSuffix $url.Path ".gif") (strings.HasSuffix $url.Path ".svg") -}} {{- $ogImage = partial "templates/og-image-url.html" $candidate -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 3. Site default OG image / logo. */ -}} {{- if not $ogImage -}} {{- with site.Params.images -}} {{- if reflect.IsSlice . -}} {{- with index . 0 -}}{{ $ogImage = partial "templates/og-image-url.html" . }}{{- end -}} {{- else -}} {{- $ogImage = partial "templates/og-image-url.html" . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* ------------------------------------------------------------------ */ -}} {{- /* Open Graph tags */ -}} {{- /* ------------------------------------------------------------------ */ -}} {{- with or site.Title site.Params.title | plainify }} {{- end }} {{- with or .Title site.Title site.Params.title | plainify }} {{- end }} {{- /* Description: front matter, then HTML-stripped summary, then site. */ -}} {{- with or .Description (.Summary | plainify) site.Params.description | plainify | htmlUnescape }} {{- end }} {{- with or .Params.locale site.Language.LanguageCode }} {{- end }} {{- /* Non-post pages (home, sections, taxonomy, ...) are websites. */ -}} {{- $ogType := "website" -}} {{- if and .IsPage (eq .Section "posts") -}} {{- $ogType = "article" -}} {{- if eq .Params.og_type "website" -}}{{ $ogType = "website" }}{{- end -}} {{- else if eq .Params.og_type "article" -}} {{- $ogType = "article" -}} {{- end }} {{- if eq $ogType "article" }} {{- with .Section }} {{- end }} {{- $ISO8601 := "2006-01-02T15:04:05-07:00" }} {{- with .PublishDate }} {{- end }} {{- with .Lastmod }} {{- end }} {{- range .GetTerms "tags" | first 6 }} {{- end }} {{- end }} {{- with $ogImage }} {{- end }} {{- /* Site default logo. NOTE: `og:logo` is NOT part of the Open Graph protocol and no crawler reads it (Facebook, LinkedIn and X all ignore it). It is emitted ONLY to satisfy a site audit that checks for the tag's presence. The logo that consumers actually read is the Schema.org Organization/BlogPosting `logo` property in templates/schema_json.html, and both point at the same asset so there is a single source of truth. Delete this block if the audit is ever fixed. */ -}} {{- with site.Params.assets.favicon | default "favicon.ico" | absURL }} {{- end }} {{- with .Params.audio }} {{- range . | first 6 }} {{- end }} {{- end }} {{- with .Params.videos }} {{- range . | first 6 }} {{- end }} {{- end }} {{- range .GetTerms "series" }} {{- range .Pages | first 7 }} {{- if ne $ . }} {{- end }} {{- end }} {{- end }} {{- with site.Params.social }} {{- if reflect.IsMap . }} {{- with .facebook_app_id }} {{- else }} {{- with .facebook_admin }} {{- end }} {{- end }} {{- end }} {{- end }} {{- with (.Param "social.fediverse_creator") }} {{- end }}
